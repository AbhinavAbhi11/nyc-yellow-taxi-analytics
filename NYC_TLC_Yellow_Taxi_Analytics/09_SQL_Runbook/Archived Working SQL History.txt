DROP TABLE IF EXISTS stg_yellow_taxi;

CREATE TABLE stg_yellow_taxi AS
SELECT
    r.*,

    /* =========================
       BASIC DATA QUALITY FLAGS
       ========================= */

    CASE
        WHEN passenger_count = 0 THEN 1
        ELSE 0
    END AS zero_passenger_flag,

    CASE
        WHEN trip_distance = 0 THEN 1
        ELSE 0
    END AS zero_distance_flag,

    CASE
        WHEN fare_amount < 0 THEN 1
        ELSE 0
    END AS negative_fare_flag,

    CASE
        WHEN total_amount < 0 THEN 1
        ELSE 0
    END AS negative_total_flag,

    CASE
        WHEN ratecode_id = 99 THEN 1
        ELSE 0
    END AS unknown_ratecode_flag,

    CASE
        WHEN pickup_datetime IS NULL
          OR dropoff_datetime IS NULL
        THEN 1
        ELSE 0
    END AS missing_timestamp_flag,

    /* =========================
       DURATION QUALITY
       ========================= */

    CASE
        WHEN pickup_datetime IS NOT NULL
         AND dropoff_datetime IS NOT NULL
         AND TIMESTAMPDIFF(
                SECOND,
                pickup_datetime,
                dropoff_datetime
             ) > 120 * 60
        THEN 1
        ELSE 0
    END AS duration_anomaly_flag,

    /* =========================
       FINANCIAL RECONCILIATION
       ========================= */

    ROUND(
        COALESCE(fare_amount, 0)
        + COALESCE(extra, 0)
        + COALESCE(mta_tax, 0)
        + COALESCE(tip_amount, 0)
        + COALESCE(tolls_amount, 0)
        + COALESCE(improvement_surcharge, 0)
        + COALESCE(congestion_surcharge, 0)
        + COALESCE(airport_fee, 0)
        + COALESCE(cbd_congestion_fee, 0)
        - COALESCE(total_amount, 0),
        2
    ) AS financial_difference,

    CASE
        WHEN ABS(
            ROUND(
                COALESCE(fare_amount, 0)
                + COALESCE(extra, 0)
                + COALESCE(mta_tax, 0)
                + COALESCE(tip_amount, 0)
                + COALESCE(tolls_amount, 0)
                + COALESCE(improvement_surcharge, 0)
                + COALESCE(congestion_surcharge, 0)
                + COALESCE(airport_fee, 0)
                + COALESCE(cbd_congestion_fee, 0)
                - COALESCE(total_amount, 0),
                2
            )
        ) <= 0.01
        THEN 'RECONCILES'

        WHEN ROUND(
            COALESCE(fare_amount, 0)
            + COALESCE(extra, 0)
            + COALESCE(mta_tax, 0)
            + COALESCE(tip_amount, 0)
            + COALESCE(tolls_amount, 0)
            + COALESCE(improvement_surcharge, 0)
            + COALESCE(congestion_surcharge, 0)
            + COALESCE(airport_fee, 0)
            + COALESCE(cbd_congestion_fee, 0)
            - COALESCE(total_amount, 0),
            2
        ) IN (
            3.25,
            2.50,
            5.00,
            1.75,
            4.25,
            0.75
        )
        THEN 'STRUCTURED_FEE_DIFFERENCE'

CREATE TABLE analytics_yellow_taxi AS
SELECT
    s.*,

    /* =========================
       TIME DIMENSIONS
       ========================= */

    DATE(pickup_datetime) AS pickup_date,

    DATE(dropoff_datetime) AS dropoff_date,

    HOUR(pickup_datetime) AS pickup_hour,

    DAYOFWEEK(pickup_datetime) AS pickup_day_of_week,

    DAYNAME(pickup_datetime) AS pickup_day_name,

    MONTH(pickup_datetime) AS pickup_month,

    /* =========================
       TRIP DURATION
       ========================= */

    ROUND(
        TIMESTAMPDIFF(
            SECOND,
            pickup_datetime,
            dropoff_datetime
        ) / 60.0,
        2
    ) AS trip_duration_minutes,

    /* =========================
       BUSINESS METRICS
       ========================= */

    CASE
        WHEN trip_distance > 0
        THEN ROUND(total_amount / trip_distance, 2)
        ELSE NULL
    END AS revenue_per_mile,

    CASE
        WHEN TIMESTAMPDIFF(
            SECOND,
            pickup_datetime,
            dropoff_datetime
        ) > 0
        THEN ROUND(
            total_amount /
            (
                TIMESTAMPDIFF(
                    SECOND,
                    pickup_datetime,
                    dropoff_datetime
                ) / 3600.0
            ),
            2
        )
        ELSE NULL
    END AS revenue_per_hour,

    /* =========================
       ANALYTICS QUALITY STATUS
       ========================= */

    CASE
        WHEN
            zero_passenger_flag = 0
            AND zero_distance_flag = 0
            AND negative_fare_flag = 0
            AND negative_total_flag = 0
            AND duration_anomaly_flag = 0
            AND financial_reconciliation_status = 'RECONCILES'
        THEN 'CLEAN'

        ELSE 'FLAGGED'
    END AS analytics_quality_status

FROM stg_yellow_taxi AS s;

SELECT
    vendor_id,
    pickup_datetime,
    dropoff_datetime,
    passenger_count,
    trip_distance,
    ratecode_id,
    store_and_fwd_flag,
    pickup_location_id,
    dropoff_location_id,
    payment_type,
    fare_amount,
    extra,
    mta_tax,
    tip_amount,
    tolls_amount,
    improvement_surcharge,
    total_amount,
    congestion_surcharge,
    airport_fee,
    cbd_congestion_fee,

    /* Data quality flags */
    CASE
        WHEN passenger_count = 0 THEN 1
        ELSE 0
    END AS zero_passenger_flag,

    CASE
        WHEN trip_distance = 0 THEN 1
        ELSE 0
    END AS zero_distance_flag,

    CASE
        WHEN fare_amount < 0 THEN 1
        ELSE 0
    END AS negative_fare_flag,

    CASE
        WHEN total_amount < 0 THEN 1
        ELSE 0
    END AS negative_total_flag,

    CASE
        WHEN ratecode_id NOT IN (1, 2, 3, 4, 5, 6, 99)
             OR ratecode_id IS NULL
        THEN 1
        ELSE 0
    END AS unknown_ratecode_flag,

    CASE
        WHEN pickup_datetime IS NULL
          OR dropoff_datetime IS NULL
        THEN 1
        ELSE 0
    END AS missing_timestamp_flag,

    CASE
        WHEN pickup_datetime IS NOT NULL
         AND dropoff_datetime IS NOT NULL
         AND (
              TIMESTAMPDIFF(
                  MINUTE,
                  pickup_datetime,
                  dropoff_datetime
              ) <= 0
              OR
              TIMESTAMPDIFF(
                  MINUTE,
                  pickup_datetime,
                  dropoff_datetime
              ) > 1440
         )
        THEN 1
        ELSE 0
    END AS duration_anomaly_flag,

    /* Financial reconciliation */
    ROUND(
        (
            COALESCE(fare_amount, 0)
          + COALESCE(extra, 0)
          + COALESCE(mta_tax, 0)
          + COALESCE(tip_amount, 0)
          + COALESCE(tolls_amount, 0)
          + COALESCE(improvement_surcharge, 0)
          + COALESCE(congestion_surcharge, 0)
          + COALESCE(airport_fee, 0)
          + COALESCE(cbd_congestion_fee, 0)
        )
        - COALESCE(total_amount, 0),
        2
    ) AS financial_difference,

    CASE

        /* Exact reconciliation */
        WHEN ABS(
            (
                COALESCE(fare_amount, 0)
              + COALESCE(extra, 0)
              + COALESCE(mta_tax, 0)
              + COALESCE(tip_amount, 0)
              + COALESCE(tolls_amount, 0)
              + COALESCE(improvement_surcharge, 0)
              + COALESCE(congestion_surcharge, 0)
              + COALESCE(airport_fee, 0)
              + COALESCE(cbd_congestion_fee, 0)
            )
            - COALESCE(total_amount, 0)
        ) < 0.01
        THEN 'RECONCILES'

        /* Difference explained by known fee structures */
        WHEN ABS(
            ABS(
                (
                    COALESCE(fare_amount, 0)
                  + COALESCE(extra, 0)
                  + COALESCE(mta_tax, 0)
                  + COALESCE(tip_amount, 0)
                  + COALESCE(tolls_amount, 0)
                  + COALESCE(improvement_surcharge, 0)
                  + COALESCE(congestion_surcharge, 0)
                  + COALESCE(airport_fee, 0)
                  + COALESCE(cbd_congestion_fee, 0)
                )
                - COALESCE(total_amount, 0)
            )
            -
            ABS(
                COALESCE(congestion_surcharge, 0)
              + COALESCE(airport_fee, 0)
              + COALESCE(cbd_congestion_fee, 0)
            )
        ) < 0.01
        THEN 'STRUCTURED_FEE_DIFFERENCE'

        ELSE 'UNEXPLAINED_DIFFERENCE'

    END AS financial_reconciliation_status

FROM raw_yellow_taxi;

    pickup_datetime,
    dropoff_datetime,
    passenger_count,
    trip_distance,
    ratecode_id,
    store_and_fwd_flag,
    pickup_location_id,
    dropoff_location_id,
    payment_type,
    fare_amount,
    extra,
    mta_tax,
    tip_amount,
    tolls_amount,
    improvement_surcharge,
    total_amount,
    congestion_surcharge,
    airport_fee,
    cbd_congestion_fee,

    CASE
        WHEN passenger_count = 0 THEN 1
        ELSE 0
    END,

    CASE
        WHEN trip_distance = 0 THEN 1
        ELSE 0
    END,

    CASE
        WHEN fare_amount < 0 THEN 1
        ELSE 0
    END,

    CASE
        WHEN total_amount < 0 THEN 1
        ELSE 0
    END,

    CASE
        WHEN ratecode_id IS NULL
             OR ratecode_id NOT IN (1,2,3,4,5,6,99)
        THEN 1
        ELSE 0
    END,

    CASE
        WHEN pickup_datetime IS NULL
             OR dropoff_datetime IS NULL
        THEN 1
        ELSE 0
    END,

    CASE
        WHEN pickup_datetime IS NOT NULL
         AND dropoff_datetime IS NOT NULL
         AND TIMESTAMPDIFF(MINUTE,pickup_datetime,dropoff_datetime) < 1
        THEN 1
        ELSE 0
    END,

    (
        COALESCE(fare_amount,0)
        + COALESCE(extra,0)
        + COALESCE(mta_tax,0)
        + COALESCE(tip_amount,0)
        + COALESCE(tolls_amount,0)
        + COALESCE(improvement_surcharge,0)
        + COALESCE(congestion_surcharge,0)
        + COALESCE(airport_fee,0)
        + COALESCE(cbd_congestion_fee,0)
        - COALESCE(total_amount,0)
    ),

    CASE
        WHEN ABS(
            (
                COALESCE(fare_amount,0)
                + COALESCE(extra,0)
                + COALESCE(mta_tax,0)
                + COALESCE(tip_amount,0)
                + COALESCE(tolls_amount,0)
                + COALESCE(improvement_surcharge,0)
                + COALESCE(congestion_surcharge,0)
                + COALESCE(airport_fee,0)
                + COALESCE(cbd_congestion_fee,0)
            )
            - COALESCE(total_amount,0)
        ) < 0.01
        THEN 'RECONCILES'

        WHEN ABS(
            COALESCE(congestion_surcharge,0)
            + COALESCE(airport_fee,0)
            + COALESCE(cbd_congestion_fee,0)
        ) > 0
        THEN 'STRUCTURED_FEE_DIFFERENCE'

        ELSE 'UNEXPLAINED_DIFFERENCE'
    END

        WHEN pickup_hour BETWEEN 0 AND 5 THEN 'OVERNIGHT'
        WHEN pickup_hour BETWEEN 6 AND 9 THEN 'MORNING'
        WHEN pickup_hour BETWEEN 10 AND 14 THEN 'MIDDAY'
        WHEN pickup_hour BETWEEN 15 AND 19 THEN 'EVENING'
        WHEN pickup_hour BETWEEN 20 AND 23 THEN 'NIGHT'
    END AS time_period,

    SUM(total_trips) AS total_trips,

    ROUND(SUM(total_revenue), 2) AS total_revenue,

    ROUND(
        SUM(total_revenue) / SUM(total_trips),
        2
    ) AS avg_revenue_per_trip,

    ROUND(
        SUM(total_duration_minutes) / SUM(total_trips),
        2
    ) AS avg_trip_duration_minutes

    CASE
        WHEN pickup_hour BETWEEN 0 AND 5 THEN 'OVERNIGHT'
        WHEN pickup_hour BETWEEN 6 AND 9 THEN 'MORNING'
        WHEN pickup_hour BETWEEN 10 AND 14 THEN 'MIDDAY'
        WHEN pickup_hour BETWEEN 15 AND 19 THEN 'EVENING'
        WHEN pickup_hour BETWEEN 20 AND 23 THEN 'NIGHT'
    END

ORDER BY
    CASE
        WHEN time_period = 'OVERNIGHT' THEN 1
        WHEN time_period = 'MORNING' THEN 2
        WHEN time_period = 'MIDDAY' THEN 3
        WHEN time_period = 'EVENING' THEN 4
        WHEN time_period = 'NIGHT' THEN 5
    END;

ranked_zones AS (
    SELECT
        *,
        
        NTILE(4) OVER (
            ORDER BY total_trips DESC
        ) AS volume_quartile,

        NTILE(4) OVER (
            ORDER BY avg_revenue_per_trip DESC
        ) AS value_quartile,

        NTILE(4) OVER (
            ORDER BY revenue_per_mile DESC
        ) AS efficiency_quartile

    FROM zone_performance
)

SELECT
    Borough,
    Zone,
    total_trips,
    total_revenue,
    avg_revenue_per_trip,
    avg_distance,
    avg_duration,
    revenue_per_mile,

    CASE
        WHEN volume_quartile = 1
             AND value_quartile = 1
             AND efficiency_quartile = 1
            THEN 'HIGH PRIORITY'

        WHEN volume_quartile = 1
             AND value_quartile <= 2
            THEN 'VOLUME LEADER'

        WHEN value_quartile = 1
             AND efficiency_quartile <= 2
            THEN 'HIGH VALUE'

        WHEN volume_quartile = 1
             AND efficiency_quartile >= 3
            THEN 'HIGH VOLUME / LOW EFFICIENCY'

        WHEN efficiency_quartile = 1
             AND volume_quartile >= 3
            THEN 'HIGH EFFICIENCY / LOW VOLUME'

        ELSE 'STANDARD'
    END AS location_category

FROM ranked_zones

WHERE total_trips >= 5000

ORDER BY
    CASE
        WHEN volume_quartile = 1
             AND value_quartile = 1
             AND efficiency_quartile = 1
            THEN 1
        WHEN volume_quartile = 1
             AND value_quartile <= 2
            THEN 2
        WHEN value_quartile = 1
             AND efficiency_quartile <= 2
            THEN 3
        WHEN volume_quartile = 1
             AND efficiency_quartile >= 3
            THEN 4
        WHEN efficiency_quartile = 1
             AND volume_quartile >= 3

        ELSE 6

DROP TABLE IF EXISTS route_directionality;

CREATE TABLE route_directionality AS
SELECT
    r.pickup_location_id AS location_a,
    r.dropoff_location_id AS location_b,

    r.total_trips AS trips_a_to_b,
    reverse_r.total_trips AS trips_b_to_a,

    r.total_revenue AS revenue_a_to_b,
    reverse_r.total_revenue AS revenue_b_to_a,

    ROUND(
        ABS(r.total_trips - reverse_r.total_trips),
        0
    ) AS trip_imbalance,

    ROUND(
        100 * ABS(r.total_trips - reverse_r.total_trips)
        / NULLIF(
            r.total_trips + reverse_r.total_trips,
            0
        ),
        2
    ) AS imbalance_pct

FROM route_summary r

JOIN route_summary reverse_r
    ON r.pickup_location_id = reverse_r.dropoff_location_id
    AND r.dropoff_location_id = reverse_r.pickup_location_id

WHERE r.pickup_location_id < r.dropoff_location_id;



SELECT
    COUNT(*) AS total_bidirectional_routes,
    SUM(trips_a_to_b + trips_b_to_a) AS total_trips_analyzed,
    ROUND(AVG(imbalance_pct), 2) AS avg_imbalance_pct,
    MAX(imbalance_pct) AS max_imbalance_pct
FROM route_directionality;



DROP TABLE IF EXISTS route_directionality_analysis;

CREATE TABLE route_directionality_analysis AS
SELECT
    r.location_a,
    a.Borough AS borough_a,
    a.Zone AS zone_a,

    r.location_b,
    b.Borough AS borough_b,
    b.Zone AS zone_b,

    r.trips_a_to_b,
    r.trips_b_to_a,

    r.trip_imbalance,
    r.imbalance_pct,

    r.revenue_a_to_b,
    r.revenue_b_to_a,

    ROUND(
        r.revenue_a_to_b - r.revenue_b_to_a,
        2
    ) AS revenue_difference,

    CASE
        WHEN r.trips_a_to_b > r.trips_b_to_a
            THEN CONCAT(a.Zone, ' → ', b.Zone)

        WHEN r.trips_b_to_a > r.trips_a_to_b
            THEN CONCAT(b.Zone, ' → ', a.Zone)

        ELSE 'BALANCED'
    END AS dominant_direction

FROM route_directionality r

LEFT JOIN taxi_zone_lookup a
    ON r.location_a = a.LocationID

LEFT JOIN taxi_zone_lookup b
    ON r.location_b = b.LocationID;
    
    
    
    SELECT
    zone_a,
    zone_b,

    trips_a_to_b,
    trips_b_to_a,

    trip_imbalance,
    imbalance_pct,

    dominant_direction,

    revenue_a_to_b,
    revenue_b_to_a,
    revenue_difference

FROM route_directionality_analysis

WHERE
    trips_a_to_b + trips_b_to_a >= 500

ORDER BY trip_imbalance DESC

LIMIT 25;

    combined_revenue,

    CASE
        WHEN imbalance_pct >= 40

            THEN 'CRITICAL_DIRECTIONAL_OPPORTUNITY'

             AND combined_revenue >= 100000
            THEN 'HIGH_DIRECTIONAL_OPPORTUNITY'

            THEN 'MODERATE_DIRECTIONAL_OPPORTUNITY'

        ELSE 'BALANCED'
    END AS route_priority

CREATE TABLE route_strategy AS
SELECT
    location_a,
    borough_a,
    zone_a,

    location_b,
    borough_b,
    zone_b,

    trips_a_to_b,
    trips_b_to_a,

    trip_imbalance,
    imbalance_pct,

    dominant_direction,

    revenue_a_to_b,
    revenue_b_to_a,

    revenue_difference,

    ROUND(
        COALESCE(revenue_a_to_b, 0) +
        COALESCE(revenue_b_to_a, 0),
        2
    ) AS combined_revenue,

    CASE

        WHEN imbalance_pct >= 40
             AND (
                 COALESCE(revenue_a_to_b, 0) +
                 COALESCE(revenue_b_to_a, 0)
             ) >= 200000
            THEN 'CRITICAL_DIRECTIONAL_OPPORTUNITY'

        WHEN imbalance_pct >= 30
             AND (
                 COALESCE(revenue_a_to_b, 0) +
                 COALESCE(revenue_b_to_a, 0)
             ) >= 100000
            THEN 'HIGH_DIRECTIONAL_OPPORTUNITY'

        WHEN imbalance_pct >= 20
            THEN 'MODERATE_DIRECTIONAL_OPPORTUNITY'

        ELSE 'BALANCED'

    END AS route_priority

FROM route_directionality_analysis;

        WHEN 'HIGH_DIRECTIONAL_OPPORTUNITY' THEN 2

DROP TABLE IF EXISTS route_performance;

CREATE TABLE route_performance AS
SELECT
    r.pickup_location_id,
    r.pickup_borough,
    r.pickup_zone,
    r.dropoff_location_id,
    r.dropoff_borough,
    r.dropoff_zone,

    r.total_trips,
    r.total_revenue,
    r.avg_revenue_per_trip,
    r.avg_distance,
    r.avg_duration,
    r.revenue_per_mile,

    COALESCE(d.trips_a_to_b, 0) AS trips_a_to_b,
    COALESCE(d.trips_b_to_a, 0) AS trips_b_to_a,

    COALESCE(d.trip_imbalance, 0) AS trip_imbalance,
    COALESCE(d.imbalance_pct, 0) AS imbalance_pct,

    COALESCE(d.revenue_a_to_b, 0) AS revenue_a_to_b,
    COALESCE(d.revenue_b_to_a, 0) AS revenue_b_to_a,

    COALESCE(
        d.revenue_a_to_b + d.revenue_b_to_a,
        r.total_revenue
    ) AS combined_route_revenue,

    COALESCE(d.revenue_difference, 0) AS revenue_difference,

    d.dominant_direction,

    CASE
        WHEN d.imbalance_pct >= 50
             AND (d.revenue_a_to_b + d.revenue_b_to_a) >= 200000
            THEN 'CRITICAL_DIRECTIONAL_OPPORTUNITY'

        WHEN d.imbalance_pct >= 30
             AND (d.revenue_a_to_b + d.revenue_b_to_a) >= 100000
            THEN 'HIGH_DIRECTIONAL_OPPORTUNITY'

        WHEN d.imbalance_pct >= 20
            THEN 'MODERATE_DIRECTIONAL_OPPORTUNITY'

        ELSE 'BALANCED'
    END AS route_priority

FROM route_intelligence r

LEFT JOIN route_directionality_analysis d
    ON r.pickup_location_id = d.location_a
   AND r.dropoff_location_id = d.location_b;

CREATE TABLE business_daily_summary AS
SELECT
    pickup_date,
    pickup_day_name AS day_name,
    COUNT(*) AS total_trips,
    ROUND(SUM(total_amount), 2) AS total_revenue,
    ROUND(AVG(total_amount), 2) AS avg_trip_revenue,
    ROUND(AVG(trip_distance), 2) AS avg_distance,
    ROUND(AVG(trip_duration_minutes), 2) AS avg_duration,
    ROUND(
        SUM(total_amount) / NULLIF(SUM(trip_distance), 0),
        2
    ) AS revenue_per_mile
FROM analytics_yellow_taxi_business
GROUP BY
    pickup_date,
    pickup_day_name
ORDER BY
    pickup_date;
    
    
    
    
SELECT
    COUNT(*) AS days,
    MIN(pickup_date) AS first_date,
    MAX(pickup_date) AS last_date,
    SUM(total_trips) AS total_trips,
    ROUND(SUM(total_revenue), 2) AS total_revenue
FROM business_daily_summary;
